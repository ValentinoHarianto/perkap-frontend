type ButtonProps = {
  children: ReactNode; variant?: "primary" | "secondary" | "danger" | "warning" | "ghost";
  icon?: IconName; onClick?: () => void; type?: "button" | "submit"; className?: string; disabled?: boolean;
};

function Button(props: ButtonProps) {
  const { children, variant = "primary", icon, onClick, type = "button", className = "" } = props;
  return <button type={type} className={`btn btn--${variant} ${className}`} onClick={onClick} disabled={props.disabled ?? false}>{icon && <Icon name={icon} size={17} />}<span>{children}</span></button>;
}

function Field({ label, placeholder, type = "text", icon, value, onChange }: { label: string; placeholder: string; type?: string; icon?: IconName; value?: string; onChange?: (value: string) => void }) {

}

function downloadText(filename: string, content: string) {
  const url = URL.createObjectURL(new Blob([content], { type: "text/plain" }));
  const link = document.createElement("a");
  link.href = url;
  link.download = filename;
  link.click();
  URL.revokeObjectURL(url);
}

function StudentRequestForm({ service, onBack, onSubmit }: { service: string; onBack: () => void; onSubmit: () => void }) {
  const [file, setFile] = useState<File | null>(null);
  const [error, setError] = useState("");
  const [showCalendar, setShowCalendar] = useState(false);
  const item = services.find((entry) => entry.title === service) ?? services[0];
  const validateFile = (selected?: File) => {
    if (!selected) return;
    if (selected.type !== "application/pdf" || !selected.name.toLowerCase().endsWith(".pdf")) {
      setFile(null);
      setError("Format tidak didukung. Surat Permohonan Peminjaman wajib berupa PDF (.pdf).");
      return;
    }
    if (selected.size > 5 * 1024 * 1024) {
      setFile(null);
      setError("Ukuran file melebihi batas maksimum 5 MB.");
      return;
    }
    setFile(selected);
    setError("");
  };
  return <>
    <div className="backline"><button onClick={onBack}><Icon name="arrow" size={17} /> Kembali ke katalog</button><Badge tone="purple">FORMULIR PEMINJAMAN</Badge></div>

      <div className="form-grid"><Field label="Tanggal pinjam" placeholder="18 Maret 2026" /><Field label="Tanggal kembali" placeholder="20 Maret 2026" /></div>

function AvailabilityCalendarModal({ service, onClose }: { service: string; onClose: () => void }) {
  const [month, setMonth] = useState(new Date(2026, 2, 1));
  const item = services.find((entry) => entry.title === service) ?? services[0];
  const monthNames = ["Januari", "Februari", "Maret", "April", "Mei", "Juni", "Juli", "Agustus", "September", "Oktober", "November", "Desember"];
  const weekdays = ["Sen", "Sel", "Rab", "Kam", "Jum", "Sab", "Min"];
  const year = month.getFullYear();
  const monthIndex = month.getMonth();
  const daysInMonth = new Date(year, monthIndex + 1, 0).getDate();
  const firstDay = (new Date(year, monthIndex, 1).getDay() + 6) % 7;
  const cells = Array.from({ length: 42 }, (_, index) => {
    const day = index - firstDay + 1;
    if (day < 1 || day > daysInMonth) return null;
    const selected = year === 2026 && monthIndex === 2 && day >= 18 && day <= 20;
    const unavailable = item.available === 0 || (!selected && (day % 9 === 0 || day % 13 === 0));
    const limited = !selected && !unavailable && (day % 5 === 0 || day % 7 === 0);
    return { day, selected, unavailable, limited };
  });
  const moveMonth = (offset: number) => setMonth(new Date(year, monthIndex + offset, 1));
  return <div className="modal-backdrop" role="presentation"><section className="modal availability-modal" role="dialog" aria-modal="true" aria-labelledby="availability-title">
    <div className="modal__head"><div className="modal__icon"><Icon name="calendar" /></div><div><h2 id="availability-title">Kalender ketersediaan</h2><p>{item.title} · {item.available} dari {item.total} unit tersedia saat ini</p></div><button className="icon-btn" onClick={onClose} aria-label="Tutup kalender"><Icon name="close" /></button></div>
    <div className="availability-calendar">
      <div className="calendar-toolbar"><button className="icon-btn calendar-prev" onClick={() => moveMonth(-1)} aria-label="Bulan sebelumnya"><Icon name="chevron" size={18} /></button><div><span>JADWAL RESERVASI</span><h3>{monthNames[monthIndex]} {year}</h3></div><button className="icon-btn" onClick={() => moveMonth(1)} aria-label="Bulan berikutnya"><Icon name="chevron" size={18} /></button></div>
      <div className="calendar-weekdays">{weekdays.map((day) => <span key={day}>{day}</span>)}</div>
      <div className="calendar-grid">{cells.map((cell, index) => <div key={index} className={`calendar-day ${!cell ? "empty" : ""} ${cell?.selected ? "selected" : ""} ${cell?.unavailable ? "unavailable" : ""} ${cell?.limited ? "limited" : ""}`}>{cell && <><strong>{cell.day}</strong><span>{cell.selected ? "Dipilih" : cell.unavailable ? "Penuh" : cell.limited ? function LiveCountdown({ initialSeconds, label }: { initialSeconds: number; label: string }) {
  const [seconds, setSeconds] = useState(initialSeconds);
  useEffect(() => {
    const timer = window.setInterval(() => setSeconds((current) => current > 0 ? current - 1 : 0), 1000);
    return () => window.clearInterval(timer);
  }, []);
  const hours = Math.floor(seconds / 3600).toString().padStart(2, "0");
  const minutes = Math.floor((seconds % 3600) / 60).toString().padStart(2, "0");
  const remaining = (seconds % 60).toString().padStart(2, "0");
  return <span className="live-countdown"><i /><strong>{hours}:{minutes}:{remaining}</strong><small>{label}</small></span>;
}

type RequestRow = { id: string; initials: string; name: string; nim: string; type: string; date: string; status: string };
type LiveSubmission = { id: string; email: string; service: string; file: string; submittedAt: string; status: string };
type MaintenanceItem = { serial: string; item: string; issue: string; photo: string; reportedAt: string; status: "In Maintenance" | "Inspection" };
type PhysicalUnitStatus = "Tersedia" | "Direservasi" | "Dipinjam" | "Perawatan";

function getPhysicalUnitStatus(item: typeof services[number], serial: string, available: number, maintenanceItems: MaintenanceItem[] = [], liveSubmission?: LiveSubmission, studentSubmitted = false): PhysicalUnitStatus {
  if (maintenanceItems.some((entry) => entry.serial === serial)) return "Perawatan";
  const liveIsActive = Boolean(studentSubmitted && liveSubmission && liveSubmission.service === item.title && ["Pending Review", "Disetujui", "Dipinjam", "In Maintenance"].includes(liveSubmission.status));
  const liveSerial = item.serials[0];
  if (liveIsActive && liveSubmission && serial === liveSerial) {
    if (liveSubmission.status === "In Maintenance") return "Perawatan";
    if (liveSubmission.status === "Dipinjam") return "Dipinjam";
    return "Direservasi";
  }
  const excluded = new Set(maintenanceItems.filter((entry) => entry.item === item.title).map((entry) => entry.serial));
  if (liveIsActive) excluded.add(liveSerial);
  const availableSerials = item.serials.filter((entry) => !excluded.has(entry)).slice(0, available);
  return availableSerials.includes(serial) ? "Tersedia" : "Dipinjam";
}

function PhysicalStatusBadge({ item, serial, available, maintenanceItems = [], liveSubmission, studentSubmitted = false }: { item: typeof services[number]; serial: string; available: number; maintenanceItems?: MaintenanceItem[]; liveSubmission?: LiveSubmission; studentSubmitted?: boolean }) {
  const status = getPhysicalUnitStatus(item, serial, available, maintenanceItems, liveSubmission, studentSubmitted);
  return <Badge tone={status === "Tersedia" ? "green" : status === "Perawatan" ? "red" : "yellow"}>{status}</Badge>;
}

"Nonaktifkan" : "Aktifkan"}</Button><Button variant="danger" onClick={() => setUsers((current) => current.filter((entry) => entry.id !== user.id))}>Hapus</Button></div></td></tr>)}</tbody></table></div></section>{showAddUser && <AddUserModal onClose={() => setShowAddUser(false)} onAdd={(user) => { setUsers((current) => [...current, user]); setShowAddUser(false); }} />}</>;
}

function Dashboard({ role, onLogout, liveSubmission, onStudentSubmit, onLiveStatus, onStockChange, stockLevels = {}, studentSubmitted = false }: { role: Role; onLogout: () => void; liveSubmission: LiveSubmission; onStudentSubmit: (service: string) => void; onLiveStatus: (status: string) => void; onStockChange: (service: string, delta: number) => void; stockLevels?: Record<string, number>; studentSubmitted?: boolean }) {
  const roleNav: Record<Role, NavItem[]> = {
    student: [{ id: "catalog", label: "Katalog Alat", icon: "box" }, { id: "history", label: "Riwayat Peminjaman", icon: "history" }, { id: "notifications", label: "Notifikasi", icon: "bell" }],
    staff: [{ id: "validation", label: "Validasi Peminjaman", icon: "check" }, { id: "loans", label: "Peminjaman Aktif", icon: "clock" }, { id: "scanner", label: "Scan QR / Handover", icon: "scan" }, { id: "maintenance", label: "Maintenance / Perbaikan", icon: "settings" }, { id: "inventory", label: "Stok Fisik & Seri", icon: "box" }, { id: "reports", label: "Laporan", icon: "chart" }],
    admin: [{ id: "users", label: "Kelola User", icon: "users" }, { id: "services", label: "Master Alat Perkap", icon: "settings" }, { id: "audit", label: "Log Aktivitas & Audit Trail", icon: "history" }],
  };
  const [active, setActive] = useState(role === "student" ? "catalog" : role === "staff" ? "validation" : "users");
  useEffect(() => setActive(role === "student" ? "catalog" : role === "staff" ? "validation" : "users"), [role]);
  const [selectedService, setSelectedService] = useState(services[0].title);
  const [collapsed, setCollapsed] = useState(false);
  const [mobileOpen, setMobileOpen] = useState(false);
  const [followup, setFollowup] = useState("");
  const [showSyncToast, setShowSyncToast] = useState(false);
  const [maintenanceItems, setMaintenanceItems] = useState<MaintenanceItem[]>([{ serial: "CAM-003", item: "Kamera Mirrorless Sony", issue: "Layar LCD retak setelah pengembalian", photo: "incident-cam003.jpg", reportedAt: "12 Mar · 09:18", status: "In Maintenance" }, { serial: "SND-004", item: "Sound System Portable", issue: "Suara kanal kanan terputus-putus", photo: "incident-snd004.jpg", reportedAt: "11 Mar · 16:40", status: "Inspection" }]);
  const resolveMaintenance = (serial: string) => {
    const resolved = maintenanceItems.find((item) => item.serial === serial);
    if (!resolved) return;
    setMaintenanceItems((current) => current.filter((item) => item.serial !== serial));
    if (liveSubmission.status === "In Maintenance" && resolved.item === liveSubmission.service) onLiveStatus("Dikembalikan");
    else onStockChange(resolved.item, 1);
  };
  const titles: Record<string, string> = { catalog: "Katalog Alat", request: "Ajukan Peminjaman", history: "Riwayat Peminjaman", tracking: "Status #PJM-2026-0142", notifications: "Notifikasi", validation: "Validasi Peminjaman", loans: "Peminjaman Aktif & Overdue", scanner: "Scan QR / Handover", maintenance: "Maintenance / Perbaikan", inventory: "Stok Fisik & Seri", reports: "Laporan Perkap", users: "Kelola User", services: "Master Alat Perkap", audit: "Log Aktivitas & Audit Trail" };
  const navigate = (id: string) => { setActive(id); setMobileOpen(false); };
  const studentContent = active === "request" ? <StudentRequestForm service={selectedService} onBack={() => setActive("catalog")} onSubmit={() => { onStudentSubmit(selectedService); setShowSyncToast(true); setActive("tracking"); }} /> : active === "tracking" ? <StudentTracking liveSubmitted={studentSubmitted} liveStatus={liveSubmission.status} service={studentSubmitted ? liveSubmission.service : services[0].title} onBack={() => setActive("history")} /> : active === "history" ? <StudentHistory liveSubmission={liveSubmission} studentSubmitted={studentSubmitted} onView={() => setActive("tracking")} /> : active === "notifications" ? <StudentNotifications /> : <StudentCatalog stockLevels={stockLevels} onApply={(service) => { setSelectedService(service); setActive("request"); }} />;
  const staffContent = active === "loans" ? <ActiveLoans onFollowup={setFollowup} /> : active === "scanner" ? <QRScanner service={liveSubmission.service} onLoanStatus={onLiveStatus} onMaintenanceAdd={(item) => setMaintenanceItems((current) => current.some((entry) => entry.serial === item.serial) ? current : [item, ...current])} /> : active === "maintenance" ? <MaintenancePanel items={maintenanceItems} onUpdate={(serial, status) => setMaintenanceItems((current) => current.map((item) => item.serial === serial ? { ...item, status } : item))} onResolve={resolveMaintenance} /> : active === "inventory" ? <InventoryStock stockLevels={stockLevels} maintenanceItems={maintenanceItems} liveSubmission={liveSubmission} studentSubmitted={studentSubmitted} /> : active === "reports" ? <Reports /> : <StaffValidation liveSubmission={liveSubmission} onLiveStatus={onLiveStatus} />;
  return <div className={`app-shell ${mobileOpen ? "mobile-open" : ""}`}><Sidebar role={role} active={active} items={roleNav[role]} collapsed={collapsed} onToggle={() => setCollapsed(!collapsed)} onSelect={navigate} onLogout={onLogout} /><button className="mobile-overlay" onClick={() => setMobileOpen(false)} aria-label="Tutup navigasi" /><div className="app-main"><Header eyebrow={`PERKAP / ${roleLabels[role]}`} title={titles[active]} initials={role === "student" ? "AS" : role === "staff" ? "DP" : "SA"} onMenu={() => setMobileOpen(true)} onHelp={() => { window.location.href = "mailto:perkap@unika.ac.id?subject=Bantuan Portal Perkap"; }} onNotification={() => role === "student" ? setActive("notifications") : window.alert("Tidak ada notifikasi operasional baru.")} /><main className="content">{role === "student" ? studentContent : role === "staff" ? staffContent : <AdminConsole section={active} stockLevels={stockLevels} onStockChange={onStockChange} maintenanceItems={maintenanceItems} liveSubmission={liveSubmission} studentSubmitted={studentSubmitted} />}</main></div>{showSyncToast && <section className="sync-toast" role="status"><div className="sync-toast__icon"><Icon name="check" /></div><div><strong>Pengajuan Berhasil Dikirim & Disinkronkan ke Petugas!</strong><span>Stok telah direservasi dan status diperbarui secara real-time.</span></div><button onClick={() => setShowSyncToast(false)} aria-label="Tutup notifikasi"><Icon name="close" size={18} /></button><i /></section>}{followup && <FollowupModal email={followup} onClose={() => setFollowup("")} />}</div>;
}

export default function App() {
  const [role, setRole] = useState<Role | null>(null);
  const [liveSubmission, setLiveSubmission] = useState<LiveSubmission>(defaultLiveSubmission);
  const [studentSubmitted, setStudentSubmitted] = useState(false);
  const [stockLevels, setStockLevels] = useState<Record<string, number>>(() => Object.fromEntries(services.map((service) => [service.title, service.available])));
  const adjustStock = (service: string, delta: number) => {
    const item = services.find((entry) => entry.title === service);
    setStockLevels((current) => ({ ...current, [service]: Math.min(item?.total ?? Number.MAX_SAFE_INTEGER, Math.max(0, (current[service] ?? item?.available ?? 0) + delta)) }));
  };
  const visibleLiveSubmission = liveSubmission.id === "LIVE-PJM-001" ? { ...liveSubmission, service: "Sound System Portable" } : liveSubmission;
  const submitStudentRequest = (service: string) => {
    setLiveSubmission({ id: `PJM-${Date.now().toString().slice(-6)}`, email: "mahasiswa.test@unika.ac.id", service, file: "Surat Permohonan Peminjaman.pdf", submittedAt: new Date().toLocaleTimeString("id-ID", { hour: "2-digit", minute: "2-digit" }), status: "Pending Review" });
    setStudentSubmitted(true);
    setStockLevels((current) => {
      const safeCurrent = current ?? Object.fromEntries(services.map((item) => [item.title, item.available]));
      return { ...safeCurrent, [service]: Math.max(0, (safeCurrent[service] ?? services.find((item) => item.title === service)?.available ?? 0) - 1) };
    });
  };
  const changeLiveStatus = (status: string) => {
    const unavailableStatuses = ["Pending Review", "Disetujui", "Dipinjam", "In Maintenance"];
    if (studentSubmitted) {
      const wasUnavailable = unavailableStatuses.includes(liveSubmission.status);
      const willBeUnavailable = unavailableStatuses.includes(status);
      if (wasUnavailable !== willBeUnavailable) adjustStock(liveSubmission.service, willBeUnavailable ? -1 : 1);
    }
    setLiveSubmission((current) => ({ ...current, status }));
  };
  return role ? <Dashboard role={role} onLogout={() => setRole(null)} liveSubmission={visibleLiveSubmission} onStudentSubmit={submitStudentRequest} onLiveStatus={changeLiveStatus} onStockChange={adjustStock} stockLevels={stockLevels} studentSubmitted={studentSubmitted} /> : <Login onLogin={setRole} />;
}
